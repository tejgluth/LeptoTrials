import {readFile,writeFile} from 'node:fs/promises'
import {fetchRegistryPage} from './src/utils/apiClient.ts'
const dir='audit/manual-review/2026-10-17';const rows=JSON.parse(await readFile(dir+'/source-records.json','utf8'));const existing=new Set(rows.map(r=>r.study.protocolSection.identificationModule.nctId));const web=JSON.parse(await readFile(dir+'/web-coverage.json','utf8'));const ids=web.ids.filter(id=>!existing.has(id));const url=new URL('https://clinicaltrials.gov/api/v2/studies');url.searchParams.set('query.id',ids.join(' OR '));url.searchParams.set('pageSize','1000');const data=await fetchRegistryPage(url.toString());if(data.nextPageToken)throw Error('Incomplete web fetch');const found=new Set(data.studies.map(s=>s.protocolSection.identificationModule.nctId));if(ids.some(id=>!found.has(id)))throw Error('Missing web records');for(const study of data.studies)rows.push({study,retrievedAt:new Date().toISOString(),url:'https://clinicaltrials.gov/study/'+study.protocolSection.identificationModule.nctId,webOnly:true});await writeFile(dir+'/source-records.json',JSON.stringify(rows,null,2));console.log('Retrieved',ids.length,'web-only official records')
