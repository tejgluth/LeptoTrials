import {readFile,writeFile} from 'node:fs/promises'
import {createHash} from 'node:crypto'
import {gzipSync} from 'node:zlib'
import {fetchRegistryPage} from './src/utils/apiClient.ts'
const dir='audit/manual-review/2026-10-17';const current=JSON.parse(await readFile(dir+'/source-records.json','utf8'));const ledger=JSON.parse(await readFile(dir+'/decision-ledger.json','utf8'));const prior=JSON.parse(await readFile('audit/manual-review/2026-10-10/decision-ledger.json','utf8'));const ids=[...new Set([...ledger.reviews.filter(r=>r.changed||r.updatedDisplayedSinceLastAudit||r.additionalSources).map(r=>r.nctId),...prior.reviews.filter(r=>r.overrideChanged).map(r=>r.nctId)])];const records=[];const timestamp=new Date().toISOString();for(let i=0;i<ids.length;i+=100){const u=new URL('https://clinicaltrials.gov/api/v2/studies');u.searchParams.set('pageSize','1000');u.searchParams.set('query.id',ids.slice(i,i+100).join(' OR '));const page=await fetchRegistryPage(u.toString());if(page.nextPageToken)throw Error('Incomplete refetch');for(const study of page.studies)records.push({study,retrievedAt:new Date().toISOString()});}const missing=ids.filter(id=>!records.some(r=>r.study.protocolSection.identificationModule.nctId===id));const hash=x=>createHash('sha256').update(JSON.stringify(x)).digest('hex');const changes=records.filter(r=>hash(r.study.protocolSection)!==hash(current.find(x=>x.study.protocolSection.identificationModule.nctId===r.study.protocolSection.identificationModule.nctId).study.protocolSection)).map(r=>r.study.protocolSection.identificationModule.nctId);await writeFile(dir+'/pass-1-refetched.json.gz',gzipSync(JSON.stringify(records)));await writeFile(dir+'/pass-1-refetch.json',JSON.stringify({startedAt:timestamp,completedAt:new Date().toISOString(),ids,count:records.length,missing,changedProtocolIds:changes},null,2));if(missing.length||changes.length)throw Error('Refetch requires reconciliation: '+JSON.stringify({missing,changes}));console.log('Independent second retrieval:',records.length,'records; identical protocol sections; no missing IDs.')
