import {readFile,writeFile,mkdir} from 'node:fs/promises'
import {filterTrial,filterByTumorType} from './src/utils/trialFilter.ts'
import {SUPPLEMENTAL_AUDITED_STUDY_IDS} from './src/utils/apiClient.ts'
const dir='audit/manual-review/2026-10-10';await mkdir(dir,{recursive:true})
const concepts=['leptomeningeal metastasis','leptomeningeal metastases','leptomeningeal disease','leptomeningeal carcinomatosis','neoplastic meningitis','carcinomatous meningitis','meningeal metastasis','meningeal metastases','leptomeningeal neoplasm','malignant neoplasm in the leptomeninges','lymphomatous meningitis','leukemic meningitis']
const queries=[];const records=new Map();const failures=[];const attempts=[]
async function get(url){for(let n=0;n<5;n++){try{const r=await fetch(url,{signal:AbortSignal.timeout(45000)});if(!r.ok)throw Error('HTTP '+r.status);const data=await r.json();return {data,retrievedAt:new Date().toISOString()}}catch(e){attempts.push({url,attempt:n+1,error:String(e),at:new Date().toISOString()});if(n===4)throw e;await new Promise(r=>setTimeout(r,1000*2**n))}}}
async function search(kind,value){let token;const tokens=new Set();let pages=0;const ids=new Set();do{const u=new URL('https://clinicaltrials.gov/api/v2/studies');u.searchParams.set('format','json');u.searchParams.set('pageSize','1000');u.searchParams.set(kind,value);if(token)u.searchParams.set('pageToken',token);const {data,retrievedAt}=await get(u);if(!Array.isArray(data.studies))throw Error('Missing studies array');for(const s of data.studies){const id=s.protocolSection.identificationModule.nctId;ids.add(id);records.set(id,{study:s,retrievedAt,url:`https://clinicaltrials.gov/study/${id}`})}token=data.nextPageToken;if(token&&tokens.has(token))throw Error('Repeated page token');tokens.add(token);if(++pages>=30&&token)throw Error('Incomplete pagination')}while(token);queries.push({kind,value,pages,count:ids.size,ids:[...ids],completedAt:new Date().toISOString()});console.log(kind,value,ids.size)}
for(const concept of concepts)for(const kind of ['query.term','query.cond']){try{await search(kind,concept)}catch(e){failures.push({kind,value:concept,error:String(e)})}}
const independentIds=[...records.keys()];const universe=JSON.parse(await readFile('audit/manual-review/universe.json','utf8'));const expanded=JSON.parse(await readFile('audit/manual-review/expanded-candidates.json','utf8'))
const extra=['NCT05843253','NCT05372614','NCT04739358','NCT05388877','NCT02971501','NCT02838420','NCT07612488','NCT05319873','NCT05991518','NCT03029065','NCT06492915','NCT00523939','NCT04545814','NCT02939300','NCT03875235','NCT02521051','NCT02336451','NCT00570908','NCT00191373','NCT06023277','NCT00005818','NCT01713699','NCT03633110','NCT04129515','NCT04416854','NCT05004142','NCT01712659']
const ids=[...new Set([...universe.records.map(r=>r.nctId),...expanded.records.map(r=>r.nctId),...SUPPLEMENTAL_AUDITED_STUDY_IDS,...extra])];for(let i=0;i<ids.length;i+=100){await search('query.id',ids.slice(i,i+100).join(' OR '))}
for(const id of ids)if(!records.has(id))failures.push({id,error:'Requested study absent from ID query'})
const buckets=['LUNG','BREAST','MELANOMA','GBM','OTHER_SOLID'];const baseline=[...records.entries()].map(([id,r])=>({nctId:id,include:filterTrial(r.study,null).include,buckets:buckets.filter(b=>filterByTumorType(r.study,b))}));
await writeFile(dir+'/source-records.json',JSON.stringify([...records.values()],null,2));await writeFile(dir+'/query-coverage.json',JSON.stringify({queries,independentIds,attempts,failures},null,2));await writeFile(dir+'/baseline-classifications.json',JSON.stringify(baseline,null,2));await writeFile(dir+'/baseline-universe.json',JSON.stringify(universe,null,2));await writeFile(dir+'/baseline-expanded.json',JSON.stringify(expanded,null,2));console.log('TOTAL',records.size,'FAILURES',failures.length)
